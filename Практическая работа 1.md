## Задача 6
```

file="$1"
first_line=$(head -n 1 "$file")

case "$file" in
    *.c|*.js)
        if [[ "$first_line" =~ ^[[:space:]]*(//|/\*) ]]
        then
            echo "Комментарий есть"
        else
            echo "Комментария нет"
        fi
        ;;
    *.py)
        if [[ "$first_line" =~ ^[[:space:]]*# ]]
        then
            echo "Комментарий есть"
        else
            echo "Комментария нет"
        fi
        ;;
    *)
        echo "Неподдерживаемое расширение"
        ;;
esac
```
